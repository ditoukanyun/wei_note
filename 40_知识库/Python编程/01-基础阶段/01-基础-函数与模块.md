# Python 函数与模块

> 学习如何编写可复用的代码,掌握 Python 函数的定义、参数、返回值以及模块管理

---

## 为什么要使用函数?

**函数**是组织好的、可重复使用的代码块,用于实现单一或相关联的功能。

**函数的好处:**
- ✅ 代码复用: 写一次,用多次
- ✅ 模块化: 将复杂问题分解
- ✅ 易维护: 修改一处,全局生效
- ✅ 易测试: 独立单元可单独测试

**编程名言**: "代码有很多种坏味道,重复是最坏的一种!" —— Martin Fowler

## 函数的定义与调用

### 定义函数

```python
def greet(name):
    """问候函数"""
    print(f"Hello, {name}!")

# 调用函数
greet("Alice")    # 输出: Hello, Alice!
```

**函数结构:**
```python
def 函数名(参数1, 参数2, ...):
    """文档字符串 (docstring)"""
    # 函数体
    return 返回值    # 可选
```

### 返回值

```python
# 无返回值
 def say_hello():
    print("Hello!")

result = say_hello()
print(result)        # None

# 单个返回值
def add(a, b):
    return a + b

result = add(3, 5)   # 8

# 多个返回值 (实际是返回元组)
def get_min_max(numbers):
    return min(numbers), max(numbers)

min_val, max_val = get_min_max([3, 1, 4, 1, 5])
print(min_val, max_val)   # 1 5
```

## 函数参数详解

### 1. 位置参数

```python
def power(base, exponent):
    return base ** exponent

# 按位置传参
result = power(2, 3)      # 8 (2³)
```

### 2. 关键字参数

```python
# 按名称传参
result = power(base=2, exponent=3)
result = power(exponent=3, base=2)  # 顺序无关

# 混合使用 (位置参数在前)
result = power(2, exponent=3)
```

### 3. 默认参数值

```python
def greet(name, greeting="Hello"):
    print(f"{greeting}, {name}!")

greet("Alice")                 # Hello, Alice!
greet("Bob", "Hi")             # Hi, Bob!
greet(name="Charlie")          # Hello, Charlie!
```

**⚠️ 注意: 默认参数必须是不可变类型!**

```python
# ❌ 错误示例
def add_item(item, items=[]):
    items.append(item)
    return items

print(add_item(1))    # [1]
print(add_item(2))    # [1, 2] (意外!)

# ✅ 正确做法
def add_item(item, items=None):
    if items is None:
        items = []
    items.append(item)
    return items
```

### 4. 可变参数 *args

```python
def sum_all(*args):
    """接收任意数量的位置参数"""
    result = 0
    for num in args:
        result += num
    return result

# 调用
sum_all()               # 0
sum_all(1)              # 1
sum_all(1, 2, 3)        # 6
sum_all(1, 2, 3, 4, 5)  # 15

# args 是一个元组
print(type(args))       # <class 'tuple'>
```

### 5. 关键字可变参数 **kwargs

```python
def print_info(**kwargs):
    """接收任意数量的关键字参数"""
    for key, value in kwargs.items():
        print(f"{key}: {value}")

# 调用
print_info(name="Alice", age=25, city="Beijing")
# 输出:
# name: Alice
# age: 25
# city: Beijing

# kwargs 是一个字典
print(type(kwargs))     # <class 'dict'>
```

### 6. 参数顺序规则

```python
def complex_function(pos1, pos2, default="value", *args, **kwargs):
    """
    参数顺序必须是:
    1. 位置参数
    2. 默认参数
    3. *args
    4. **kwargs
    """
    print(f"位置参数: {pos1}, {pos2}")
    print(f"默认参数: {default}")
    print(f"可变参数: {args}")
    print(f"关键字参数: {kwargs}")

complex_function(1, 2, "custom", 3, 4, 5, name="Alice", age=25)
```

## 函数进阶特性

### 匿名函数 Lambda

```python
# 语法: lambda 参数: 表达式

# 简单函数
square = lambda x: x ** 2
print(square(5))      # 25

# 多参数
add = lambda x, y: x + y
print(add(3, 5))      # 8

# 立即执行
result = (lambda x, y: x * y)(4, 5)   # 20

# 常见应用场景
numbers = [3, 1, 4, 1, 5, 9, 2, 6]

# 排序
sorted(numbers, key=lambda x: -x)     # 降序

# map 操作
list(map(lambda x: x ** 2, numbers))  # [9, 1, 16, 1, 25, 81, 4, 36]

# filter 操作
list(filter(lambda x: x > 5, numbers))  # [9, 6]
```

**何时使用 lambda?**
- 简单的一次性函数
- 作为参数传递给高阶函数

**何时使用 def?**
- 复杂逻辑
- 需要文档字符串
- 需要多次复用

### 高阶函数

**函数可以作为参数传递:**

```python
def apply_operation(x, y, operation):
    """接收函数作为参数"""
    return operation(x, y)

def add(a, b):
    return a + b

def multiply(a, b):
    return a * b

# 调用
result = apply_operation(3, 5, add)       # 8
result = apply_operation(3, 5, multiply)  # 15
result = apply_operation(3, 5, lambda x, y: x - y)  # -2
```

**常用高阶函数:**

```python
numbers = [1, 2, 3, 4, 5]

# map: 对每个元素应用函数
squares = list(map(lambda x: x**2, numbers))
# [1, 4, 9, 16, 25]

# filter: 筛选满足条件的元素
evens = list(filter(lambda x: x % 2 == 0, numbers))
# [2, 4]

# reduce: 累积计算
from functools import reduce
product = reduce(lambda x, y: x * y, numbers)
# 120 (1*2*3*4*5)

total = reduce(lambda x, y: x + y, numbers, 0)
# 15 (初始值0 + 1+2+3+4+5)

# sorted: 排序
sorted(numbers, reverse=True)   # [5, 4, 3, 2, 1]
sorted(['banana', 'apple'], key=len)  # 按长度排序
```

### 偏函数 (Partial)

```python
from functools import partial

# 固定部分参数
def power(base, exponent):
    return base ** exponent

# 创建平方函数
square = partial(power, exponent=2)
print(square(5))      # 25

# 创建立方函数
cube = partial(power, exponent=3)
print(cube(3))        # 27
```

## 装饰器 (Decorator)

### 什么是装饰器?

装饰器是一种**在不修改原函数代码**的情况下,增加额外功能的技术。

### 基础装饰器

```python
import time

def timing(func):
    """计算函数执行时间的装饰器"""
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        elapsed = time.time() - start
        print(f"{func.__name__} 执行时间: {elapsed:.4f}秒")
        return result
    return wrapper

@timing
def slow_function():
    time.sleep(1)
    print("函数执行完成")

slow_function()
# 输出:
# 函数执行完成
# slow_function 执行时间: 1.0012秒
```

### 带参数的装饰器

```python
def repeat(n):
    """重复执行 n 次的装饰器"""
    def decorator(func):
        def wrapper(*args, **kwargs):
            for _ in range(n):
                result = func(*args, **kwargs)
            return result
        return wrapper
    return decorator

@repeat(3)
def greet():
    print("Hello!")

greet()
# 输出:
# Hello!
# Hello!
# Hello!
```

### 常用装饰器示例

```python
# 1. 日志装饰器
def log(func):
    def wrapper(*args, **kwargs):
        print(f"调用 {func.__name__} 参数: {args}, {kwargs}")
        result = func(*args, **kwargs)
        print(f"返回: {result}")
        return result
    return wrapper

# 2. 缓存装饰器 (Memoization)
from functools import lru_cache

@lru_cache(maxsize=128)
def fibonacci(n):
    if n < 2:
        return n
    return fibonacci(n-1) + fibonacci(n-2)

# 3. 权限检查装饰器
def require_login(func):
    def wrapper(user, *args, **kwargs):
        if not user.is_logged_in:
            raise Exception("请先登录!")
        return func(user, *args, **kwargs)
    return wrapper
```

## 递归函数

### 什么是递归?

函数调用自身的过程称为**递归**。

```python
# 经典例子: 阶乘
def factorial(n):
    """计算 n!"""
    if n <= 1:
        return 1
    return n * factorial(n - 1)

print(factorial(5))   # 120

# 执行过程:
# factorial(5) = 5 * factorial(4)
#              = 5 * 4 * factorial(3)
#              = 5 * 4 * 3 * factorial(2)
#              = 5 * 4 * 3 * 2 * factorial(1)
#              = 5 * 4 * 3 * 2 * 1
#              = 120
```

### 递归三要素

1. **递归终止条件** (Base Case): 防止无限递归
2. **递归调用** (Recursive Case): 函数调用自身
3. **问题规模减小**: 每次递归向终止条件靠近

```python
# 斐波那契数列
def fib(n):
    if n <= 1:           # 终止条件
        return n
    return fib(n-1) + fib(n-2)   # 递归调用

# 汉诺塔
def hanoi(n, source, target, auxiliary):
    """将 n 个盘子从 source 移到 target"""
    if n == 1:
        print(f"将盘子 1 从 {source} 移到 {target}")
        return
    hanoi(n-1, source, auxiliary, target)
    print(f"将盘子 {n} 从 {source} 移到 {target}")
    hanoi(n-1, auxiliary, target, source)

hanoi(3, 'A', 'C', 'B')
```

### 尾递归优化

```python
# 普通递归 (有栈溢出风险)
def factorial(n):
    if n <= 1:
        return 1
    return n * factorial(n - 1)

# 尾递归 (Python 不优化,但思路值得学习)
def factorial_tail(n, acc=1):
    if n <= 1:
        return acc
    return factorial_tail(n - 1, n * acc)
```

## 模块管理

### 什么是模块?

模块是包含 Python 代码的文件,可以定义函数、类和变量。

**优点:**
- 代码组织
- 命名空间管理
- 代码复用

### 导入模块

```python
# 方式1: 导入整个模块
import math
print(math.pi)
print(math.sqrt(16))

# 方式2: 导入特定函数/变量
from math import pi, sqrt
print(pi)
print(sqrt(16))

# 方式3: 使用别名
import math as m
print(m.pi)

from math import sqrt as square_root
print(square_root(16))

# 方式4: 导入所有 (不推荐!)
from math import *
print(pi)
print(sqrt(16))
```

### 自定义模块

**文件: mymath.py**
```python
"""数学工具模块"""

PI = 3.14159

def add(a, b):
    return a + b

def multiply(a, b):
    return a * b

class Calculator:
    def __init__(self):
        self.history = []
    
    def calculate(self, a, b, op):
        if op == '+':
            result = add(a, b)
        elif op == '*':
            result = multiply(a, b)
        self.history.append(f"{a} {op} {b} = {result}")
        return result
```

**文件: main.py**
```python
import mymath

# 使用模块
print(mymath.PI)
print(mymath.add(3, 5))

calc = mymath.Calculator()
result = calc.calculate(10, 20, '+')
print(result)
```

## 🎯 实战案例

### 案例1: 文件处理工具函数

```python
"""文件处理工具模块"""

def read_file(filepath, encoding='utf-8'):
    """安全读取文件"""
    try:
        with open(filepath, 'r', encoding=encoding) as f:
            return f.read()
    except FileNotFoundError:
        return None
    except Exception as e:
        print(f"读取错误: {e}")
        return None

def write_file(filepath, content, encoding='utf-8'):
    """安全写入文件"""
    try:
        with open(filepath, 'w', encoding=encoding) as f:
            f.write(content)
        return True
    except Exception as e:
        print(f"写入错误: {e}")
        return False

def count_lines(filepath):
    """统计文件行数"""
    content = read_file(filepath)
    if content is None:
        return 0
    return len(content.splitlines())

# 使用
content = read_file('data.txt')
write_file('output.txt', 'Hello, World!')
print(f"总行数: {count_lines('data.txt')}")
```

### 案例2: 带缓存的 Fibonacci

```python
from functools import lru_cache

@lru_cache(maxsize=None)
def fibonacci(n):
    """带缓存的斐波那契数列"""
    if n < 2:
        return n
    return fibonacci(n-1) + fibonacci(n-2)

# 对比性能
import time

# 递归版本 (慢)
def fib_slow(n):
    if n < 2:
        return n
    return fib_slow(n-1) + fib_slow(n-2)

# 测试
start = time.time()
print(fibonacci(35))    # 很快!
print(f"耗时: {time.time() - start:.4f}秒")

start = time.time()

print(f"耗时: {time.time() - start:.4f}秒")
```

### 案例3: 函数式编程处理数据

```python
# 学生成绩数据处理
students = [
    {'name': 'Alice', 'scores': [85, 90, 78]},
    {'name': 'Bob', 'scores': [92, 88, 95]},
    {'name': 'Charlie', 'scores': [78, 85, 80]},
    {'name': 'David', 'scores': [95, 92, 98]}
]

# 计算平均分
def get_average(student):
    return sum(student['scores']) / len(student['scores'])

# 1. 计算所有学生平均分
averages = list(map(get_average, students))
print(f"平均分列表: {averages}")

# 2. 筛选优秀学生 (>90分)
excellent = list(filter(lambda s: get_average(s) > 90, students))
print(f"优秀学生: {[s['name'] for s in excellent]}")

# 3. 按平均分排序
ranked = sorted(students, key=get_average, reverse=True)
print("排名:")
for i, s in enumerate(ranked, 1):
    print(f"{i}. {s['name']}: {get_average(s):.2f}")

# 4. 计算班级总平均分
class_avg = sum(map(get_average, students)) / len(students)
print(f"班级平均分: {class_avg:.2f}")
```

## 📝 重点总结

### 函数编写最佳实践

```python
# ✅ 好的函数示例
def calculate_area(length, width):
    """
    计算矩形面积
    
    Args:
        length: 长度
        width: 宽度
    
    Returns:
        矩形面积
    """
    if length <= 0 or width <= 0:
        raise ValueError("长度和宽度必须大于0")
    return length * width

# ✅ 函数设计原则
# 1. 单一职责: 一个函数只做一件事
# 2. 见名知意: 函数名清晰表达功能
# 3. 参数精简: 参数不要太多 (<=5个)
# 4. 文档完善: 添加 docstring
# 5. 异常处理: 处理边界情况
```
